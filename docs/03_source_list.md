### 4. 備忘録：Web掲載用HTML一括生成 GASコード

スプレッドシートの「拡張機能」＞「Apps Script」に登録し、セル上で `=GET_COMPLIANCE_HTML(B2)` のように呼び出して使用するカスタム関数です。Difyの代替として、加算状況から一瞬でHTMLを出力します。

```javascript
/**
 * 加算状況からWeb掲載用のHTMLを出力するカスタム関数
 * @param {string} kasan_list - 加算状況のテキスト
 * @return {string} 出力されるHTMLテキスト
 * @customfunction
 */
function GET_COMPLIANCE_HTML(kasan_list) {
  if (!kasan_list) {
    kasan_list = "";
  }
  
  var output_html = "";
  
  // 1. 連携強化加算の判定（高リスク：第二種協定指定）
  if (kasan_list.indexOf("連携強化加算") !== -1) {
    output_html += "<h3>災害や新興感染症発生時における対応体制（連携強化加算）</h3>\n<p>当薬局は、都道府県知事から「第二種協定指定医療機関」の指定を受けており、災害や新興感染症の発生時等において対応可能な体制を確保しています。</p>\n<ul><li>自治体等からの要請に応じて、自宅療養者等に対する調剤、オンライン又は訪問による服薬指導、および薬剤の交付（配達等）を行う体制を整備しています。</li>\n<li>新型コロナウイルス等の抗原検査キットや、感染対策に必要な衛生材料等を提供できる体制を整えています。</li></ul>\n\n";
  }
  
  // 2. 電子的調剤情報連携体制整備加算の判定（名称変更対応・旧名称フォールバック）
  if (kasan_list.indexOf("電子的調剤情報連携体制整備加算") !== -1 || kasan_list.indexOf("医療DX") !== -1) {
    output_html += "<h3>電子的調剤情報連携体制について</h3>\n<p>当薬局では、医療DXを推進し質の高い医療を提供するため、以下の体制を整備しています。</p>\n<ul><li>オンライン資格確認システム等を通じて患者様の診療情報・薬剤情報等を取得し、調剤・服薬指導に活用しています。</li>\n<li>マイナ保険証の利用促進など、医療DXを通じた情報取得・活用に取り組んでいます。</li>\n<li>電子処方せんの受付や、電子カルテ情報共有サービス（今後導入予定）を活用する体制を整えています。</li></ul>\n\n";
  }

  // 3. 地域支援・医薬品供給対応体制加算の判定（名称変更対応・旧名称フォールバック）
  if (kasan_list.indexOf("地域支援・医薬品供給対応体制加算") !== -1 || kasan_list.indexOf("地域支援体制加算") !== -1) {
    output_html += "<h3>地域支援・医薬品供給対応体制について</h3>\n<p>当薬局は、地域医療に貢献するため、以下の体制を整えています。</p>\n<ul><li>夜間・休日等を含む開局時間外であっても、調剤および在宅業務に対応できる体制を整備しています。</li>\n<li>地域の医療機関や訪問看護ステーション等と連携し、在宅患者様への訪問薬剤管理指導を行っています。</li>\n<li>麻薬の調剤や、要指導医薬品・一般用医薬品の提供、健康相談等に幅広く対応しています。</li></ul>\n\n";
  }

  // 4. 在宅薬学総合体制加算の判定
  if (kasan_list.indexOf("在宅薬学総合体制加算") !== -1) {
    output_html += "<h3>在宅医療への対応について</h3>\n<p>当薬局は、在宅で療養される患者様に対する薬学的管理・指導を行う体制（在宅薬学総合体制）を整備し、地域の在宅医療を支援しています。</p>\n\n";
  }

  // 5. 無菌製剤処理加算の判定
  if (kasan_list.indexOf("無菌製剤処理加算") !== -1) {
    output_html += "<h3>高度な調剤（無菌製剤処理）への対応について</h3>\n<p>当薬局では、必要に応じて無菌的な製剤処理を行うことができる体制を整えています。</p>\n\n";
  }

  // 不足事項がない場合の処理
  if (output_html === "") {
    output_html = "<p>Webサイトに追記すべき必須掲示項目は現在ありません。</p>";
  }

  return output_html.trim();
}
